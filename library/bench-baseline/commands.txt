# === (c) CK-FA decode latency micro-bench (Llama-2-7b-hf head_dim=128) ===
# Default-on backend is ROCM_ATTN; --attention-backend ROCM_CK_FA forces CK FA.
# Llama-2-7b max_model_len=4096 (forced; do NOT set 32768 — pydantic rejects).
# CK FA splitkv does NOT support head_dim=256 -> use Llama-2-7b head_dim=128.

cd /home/aimeme/Desktop/vllm-gfx908/.emdash/worktrees/vllm-gfx908/emdash/loose-rats-clean-phm2k
env TORCH_COMPILE_DISABLE=1 HF_HUB_OFFLINE=1 PYTORCH_ROCM_ARCH=gfx908 CUDA_VISIBLE_DEVICES=0 \
  /opt/vllm-env/bin/python -m vllm.entrypoints.cli.main bench latency \
    --model /models/Llama-2-7b-hf \
    --attention-backend ROCM_CK_FA \
    --dtype float16 \
    --tensor-parallel-size 1 \
    --enforce-eager \
    --max-model-len 4096 \
    --gpu-memory-utilization 0.90 \
    --input-len 2048 \
    --output-len 64 \
    --batch-size 1 \
    --num-iters 30 \
    --num-iters-warmup 10 \
    --seed 42 \
    --trust-remote-code \
    --output-json library/bench-baseline/ckfa-latency.json
